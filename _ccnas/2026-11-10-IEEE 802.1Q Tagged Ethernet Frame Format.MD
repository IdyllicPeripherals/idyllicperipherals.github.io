<script>
    const frameNames = ["Preamble", "SFD", "Destination MAC", "Source MAC", "802.1Q Tag", "Type", "Data and Pad", "FCS"];
    const frameSizes = ["7", "1", "6", "6", "4", "2", "42-1500", "4"];
    const frameDescs = ["Syncs the receiver clock", "Marks the start of the frame", "Address of the receiver", "Address of the sender", "Identifies the VLAN", "Identifies the upper layer protocol", "Carries the payload", "CRC error check"];
    const tagNames = ["TPID", "PCP", "DEI", "VID"];
    const tagSizes = ["16 bits", "3 bits", "1 bit", "12 bits"];
    const tagDescs = ["Marks the frame as 802.1Q tagged (0x8100)", "Frame priority (CoS)", "Marks frames to drop first in congestion", "The VLAN number"];

    const pools = [
        { rowId: "frameNameRow", items: frameNames, cls: "draggable" },
        { rowId: "frameSizeRow", items: frameSizes, cls: "draggable" },
        { rowId: "frameDescRow", items: frameDescs, cls: "draggable desc" },
        { rowId: "tagNameRow", items: tagNames, cls: "draggable" },
        { rowId: "tagSizeRow", items: tagSizes, cls: "draggable" },
        { rowId: "tagDescRow", items: tagDescs, cls: "draggable desc" }
    ];

    const droppables = document.querySelectorAll('.droppable');
    const congratsMessage = document.getElementById('congratsMessage');

    function shuffle(array) {
        for (let i = array.length - 1; i > 0; i--) {
            const j = Math.floor(Math.random() * (i + 1));
            [array[i], array[j]] = [array[j], array[i]];
        }
        return array;
    }

    function placeDraggablesRandomly() {
        pools.forEach(pool => {
            const row = document.getElementById(pool.rowId);
            row.innerHTML = "";
            shuffle([...pool.items]).forEach(item => {
                const td = document.createElement("td");
                td.className = pool.cls;
                td.draggable = true;
                td.textContent = item;
                td.addEventListener('dragstart', (e) => {
                    e.dataTransfer.setData('text/plain', item);
                });
                row.appendChild(td);
            });
        });
    }

    function resetDroppableCells() {
        droppables.forEach(droppable => {
            droppable.textContent = "";
            droppable.classList.remove('correct', 'incorrect');
        });
        congratsMessage.style.visibility = "hidden";
    }

    function checkCompletion() {
        let allCorrect = true;
        droppables.forEach(droppable => {
            if (droppable.textContent === "" || !droppable.classList.contains('correct')) {
                allCorrect = false;
            }
        });
        if (allCorrect) {
            congratsMessage.style.visibility = "visible";
        }
    }

    droppables.forEach(droppable => {
        droppable.addEventListener('dragover', (e) => {
            e.preventDefault();
        });

        droppable.addEventListener('drop', (e) => {
            e.preventDefault();
            const draggedText = e.dataTransfer.getData('text/plain');
            const correctId = droppable.getAttribute('data-correct');

            if (draggedText.replace(/\s/g, "") === correctId.replace(/\s/g, "")) {
                droppable.classList.add('correct');
                droppable.textContent = draggedText;
                checkCompletion();
            } else {
                droppable.classList.add('incorrect');
                setTimeout(() => {
                    droppable.classList.remove('incorrect');
                }, 1000);
            }
        });
    });

    document.getElementById("restartButton").addEventListener("click", () => {
        resetDroppableCells();
        placeDraggablesRandomly();
    });

    placeDraggablesRandomly();
</script>
